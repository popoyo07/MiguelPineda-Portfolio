Enemy AI Script Breakdown

EnemySoundAI.cs

Purpose

This script controls how the enemy reacts to player sound, moves toward the last heard location, and creates pressure during exploration.

How It Works

  • Listens for sound events created by the player.
  • Stores the last heard position.
  • Moves the enemy toward that location to investigate.
  • Returns to patrol if the player is not found.
  • Supports the stealth/horror loop by making movement feel risky.

using System.Collections;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using UnityEngine.AI;
using UnityEngine.ProBuilder.MeshOperations;

public class EnemyNavigation : MonoBehaviour
{
    // Handles enemy pathfinding and navigation through Unity's NavMesh system
    private NavMeshAgent agent;

    // Array of waypoints used for the enemy’s patrol route
    public Transform[] waypoints;

    // Keeps track of the current waypoint index
    private int wCount;

    // Reference to the player object in the scene
    private GameObject player;

    // Used to detect if the player is making noise
    private PlayerAudio playerAudio;

    // Boolean tracking if player is producing noise
    private bool playerNoise;

    // Determines if the enemy is currently chasing the player
    private bool isChasing = false;

    // Determines if the enemy is in search mode after losing the player
    public bool isSearching = false;

    // Reference to the level canvas, used to trigger UI actions like Game Over
    public GameObject levelCanvas;

    // Stores player’s last known position when detected by sound
    private Vector3 storePlayerLastPosition;

    // Used for camera targeting during capture
    public Transform enemyHead;

    // Controls enemy’s animation states
    private Animator SusEnemy;

    // Plays enemy-related sound effects
    AudioSource monsterAudio;

    // Audio clip played when the player gets caught
    [SerializeField] AudioClip catchingSound;

    private void OnTriggerEnter(Collider other)
    {
        // If enemy collides with the player, trigger capture sequence
        if (other.transform.tag == "Player")
        {
            CapturePlayer();
        }
        else if (other.transform.tag == "EndGame")
        {
            // If enemy collides with end trigger, stop and resume patrol
            agent.isStopped = true;
            Patrol();
            agent.isStopped = false;
        }
    }

    private void CapturePlayer()
    {
        // Stop animation playback to freeze the enemy’s motion
        SusEnemy.StartPlayback();

        // Stop navigation movement
        agent.isStopped = true;

        // Play the capture sound effect once
        monsterAudio.clip = catchingSound;
        monsterAudio.loop = false;
        monsterAudio.Play();

        // Make both enemy and player look at each other
        transform.LookAt(player.transform.position);
        player.transform.LookAt(transform.position);

        // Mark player as caught for the gameplay system
        player.GetComponent().gotCaught = true;

        // If available, smoothly rotate the player’s camera to face the enemy
        Camera playerCamera = player.GetComponentInChildren();
        if (playerCamera != null && enemyHead != null)
        {
            StartCoroutine(SmoothCameraFocus(playerCamera));
        }
    }

    // Coroutine that smoothly rotates the camera toward the enemy’s head
    private IEnumerator SmoothCameraFocus(Camera cam)
    {
        float duration = 1.2f;
        float elapsed = 0f;

        while (elapsed < duration)
        {
            cam.transform.rotation = Quaternion.Slerp(cam.transform.rotation,
                Quaternion.LookRotation(enemyHead.position - cam.transform.position),
                elapsed / duration);
            elapsed += Time.deltaTime;
            yield return null;
        }
    }
}
